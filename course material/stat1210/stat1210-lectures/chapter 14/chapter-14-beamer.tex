% SLIDE 2
\begin{frame}{From Sampling Distributions to Statistical Inference}
Chapter 13 showed that a sample statistic varies from one random sample to another.
\vspace{0.2cm}

This sampling variation means that a statistic such as $\bar{x}$ or $\hat{p}$ will not usually equal the corresponding population parameter exactly.

\vspace{0.2cm}
However, the sampling distribution describes how the statistic is expected to vary. We can use this information to measure the uncertainty associated with a sample result.

\vspace{0.25cm}
\begin{block}{Central Question}
\centering
\large
What can sample data tell us about an unknown population parameter?
\end{block}

\vspace{0.15cm}
\textbf{Statistical inference} uses sample data to draw conclusions about a population while accounting for random sampling variation.
\end{frame}


% SLIDE 3
\begin{frame}{Two Main Goals of Statistical Inference}
Statistical inference has two closely related goals.

\vspace{0.2cm}
\begin{block}{Estimation}
A \textbf{confidence interval} uses sample data to estimate an unknown population parameter. It provides a range of plausible values and communicates the precision of the estimate.
\end{block}

\vspace{0.15cm}
\begin{block}{Testing a Claim}
A \textbf{hypothesis test} evaluates how compatible the observed sample data are with a specific claim about a population parameter. The evidence is assessed by comparing the observed statistic with the sampling distribution expected under that claim.
\end{block}

\vspace{0.15cm}
Both methods use sampling distributions to account for the variation that occurs from one random sample to another.
\end{frame}


% SLIDE 4
\begin{frame}{Why Begin with a Known Population Standard Deviation?}
When making inferences about a population mean, the population standard deviation $\sigma$ is usually unknown.

\vspace{0.2cm}
In this chapter, we initially assume that $\sigma$ is known. Although this situation is uncommon in practice, the assumption allows us to focus on the main ideas of statistical inference.
\begin{itemize}
  \item How sampling distributions quantify uncertainty
  \item How confidence intervals are constructed and interpreted
  \item How evidence against a population claim is evaluated
  \item How sample size and confidence level affect precision
\end{itemize}

\vspace{0.15cm}
When $\sigma$ is unknown, it must be estimated using the sample standard deviation $s$. This leads to methods based on the $t$ distribution, which will be introduced later.
\end{frame}

% SECTION 2
\section{Statistical Estimation}

% SLIDE 5
\begin{frame}{Point Estimates and Interval Estimates}
A sample statistic can be used to estimate an unknown population parameter.

\vspace{0.2cm}
For a population mean $\mu$, the natural point estimate is the sample mean
$$\bar{x}$$

\begin{itemize}
	\item A \textbf{point estimate} gives a single numerical estimate of the parameter
  	\item An \textbf{interval estimate} gives a range of plausible values for the parameter
\end{itemize}

\vspace{0.2cm}
A confidence interval has the general form
$$\text{point estimate}\ \pm\ \text{margin of error}$$

The margin of error describes the uncertainty produced by random sampling variation.
\end{frame}

% SLIDE 6
\begin{frame}{Example 14.1: Estimating a Population Mean}
\begin{block}{Question}
A hospital's pediatric ward survey selects a random sample of $n=217$ eight year old patients from the population of interest. The sample mean height is $\bar{x}=132.5$ cm. For this introductory example, assume that the population standard deviation is known to be $\sigma=10$ cm. How can the sample be used to estimate the unknown population mean height $\mu$ while accounting for random sampling variation?
\end{block}

\vspace{0.2cm}
The sample mean provides the point estimate
$$\bar{x}=132.5\text{ cm}$$
However, the point estimate alone does not describe how much $\bar{x}$ would vary across repeated random samples.
\end{frame}

% SLIDE 7
\begin{frame}{Example 14.1: Calculating the Standard Error}
For random samples of size $n$, the sampling distribution of $\bar{x}$ has centre and standard deviation 
$$\mu_{\bar{x}}=\mu \quad \text{and} \quad \sigma_{\bar{x}}=\frac{\sigma}{\sqrt{n}}$$
This standard deviation describes the typical distance between a sample mean and the population mean. In this setting, it is called the \textbf{standard error of the sample mean}.

\vspace{0.2cm}
For the height data
$$\sigma_{\bar{x}}=\frac{10}{\sqrt{217}}\approx 0.679\text{ cm}$$

Therefore, sample means from repeated random samples of $217$ patients would typically vary by about $0.679$ cm around $\mu$.
\end{frame}

% SLIDE 8
\begin{frame}{Example 14.2: Finding an Approximate Margin of Error}
Because the sample size is large, the sampling distribution of $\bar{x}$ is approximately Normal. The 68--95--99.7 rule states that approximately $95\%$ of a Normal distribution lies within two standard deviations of its mean.

\vspace{0.2cm}
Therefore, approximately $95\%$ of sample means will fall within two standard errors of the population mean.
$$\bar{x}\text{ is usually within }2\sigma_{\bar{x}}\text{ of }\mu$$

The approximate margin of error is
$$m=2\sigma_{\bar{x}}=2(0.679)\approx 1.36\text{ cm}$$

The value $2$ is being used as an approximate $95\%$ critical value. The more precise value is $1.96$.
\end{frame}

% SLIDE 9
\begin{frame}{Example 14.2: Constructing the Confidence Interval}
The approximate $95\%$ confidence interval has the form
$$\bar{x}\pm m$$
Substituting the sample mean and margin of error gives
$$132.5\pm1.36$$
The lower endpoint is
$$132.5-1.36=131.14$$
The upper endpoint is
$$132.5+1.36=133.86$$
After rounding appropriately, the approximate $95\%$ confidence interval is
$$\boxed{(131.1,\ 133.9)\text{ cm}}$$
\end{frame}

% SLIDE 10
\begin{frame}{Example 14.2: Interpreting the Confidence Interval}
\begin{block}{Interpretation}
Using this confidence interval procedure, we are approximately $95\%$ confident that the population mean height of eight year old patients in the population of interest is between $131.1$ cm and $133.9$ cm.
\end{block}

\vspace{0.2cm}
The confidence level describes the reliability of the \textbf{procedure} used to construct the interval.
\begin{itemize}
  \item The population mean $\mu$ is fixed but unknown
  \item The sample mean and confidence interval vary from sample to sample
  \item Some intervals contain $\mu$, while others do not
\end{itemize}

\vspace{0.15cm}
The interval does not state that $95\%$ of individual heights lie between $131.1$ cm and $133.9$ cm. It is an interval for the population mean, not for individual observations.
\end{frame}

% SLIDE 12
\begin{frame}{What Does 95\% Coverage Mean?}
A $95\%$ confidence procedure has a \textbf{long-run coverage rate of approximately $95\%$} when its assumptions are satisfied.

\vspace{0.2cm}
If the sampling process were repeated many times and a confidence interval were constructed from each sample, approximately $95\%$ of those intervals would contain the true population parameter.

\vspace{0.2cm}
\begin{alertblock}{What It Does Not Mean}
After one particular interval has been calculated, it is not correct to say that there is a $95\%$ probability that the fixed parameter lies inside that interval.
\end{alertblock}
\end{frame}

% SECTION 3
\section{Margin of Error and Confidence Level}

% SLIDE 13
\begin{frame}{Anatomy of a Confidence Interval}
Most confidence intervals have the form
$$\text{estimate}\ \pm\ \text{margin of error}$$
The margin of error can be separated into two components
$$\text{margin of error}=\text{critical value}\times\text{standard error}$$
Therefore, the general structure is
$$\boxed{\text{confidence interval}=\text{estimate}\pm(\text{critical value})(\text{standard error})}$$

Three ideas determine the interval:
\begin{itemize}
	\item The \textbf{estimate} determines the centre
  	\item The \textbf{standard error} describes sampling variability
 	\item The \textbf{confidence level} determines the required critical value
\end{itemize}
\end{frame}

% SLIDE 14
\begin{frame}{The Estimate: Locating the Centre}
When estimating a population mean $\mu$, the natural point estimate is the sample mean
$$\bar{x}$$
Therefore, a confidence interval for $\mu$ is centred at $\bar{x}$.

\vspace{0.2cm}
Under the random-sampling model, the sample mean is an \textbf{unbiased estimator} of the population mean
$$E(\bar{X})=\mu$$
Unbiasedness means that the sample mean is centred at $\mu$ across repeated random samples. It does not guarantee that the mean from one particular sample will be close to $\mu$.
\end{frame}

% SLIDE 15
\begin{frame}{The Standard Error: Measuring Sampling Variability}
When the population standard deviation $\sigma$ is known, the standard error of the sample mean is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}$$
The standard error describes how much the sample mean would typically vary from one random sample to another.
\begin{itemize}
  \item A smaller standard error means that $\bar{x}$ is usually closer to $\mu$
  \item A larger standard error means that sample means vary more across samples
  \item Larger random samples produce smaller standard errors
\end{itemize}
\end{frame}

% SLIDE 16
\begin{frame}{The Confidence Level and Critical Value}
The \textbf{confidence level} is the intended long-run coverage rate of the confidence interval procedure. For a confidence level $C$, the critical value $z^*$ is chosen so that the central area under the Standard Normal curve equals $C$:
$$P(-z^*<Z<z^*)=C$$

If $\alpha=1-C$
then the area in each tail is
$$\frac{\alpha}{2}$$

\vspace{0.15cm}
\begin{center}
\begin{tabular}{ccc}
\toprule
\textbf{Confidence level} & \textbf{Area in each tail} & \textbf{Critical value $z^*$} \\
\midrule
$90\%$ & $0.050$ & $1.645$ \\
$95\%$ & $0.025$ & $1.960$ \\
$99\%$ & $0.005$ & $2.576$ \\
\bottomrule
\end{tabular}
\end{center}

A higher confidence level requires a larger critical value.
\end{frame}

% SLIDE 17
\begin{frame}{The Margin of Error}
For a confidence interval for a population mean with known $\sigma$, the margin of error is
$$m=z^*\operatorname{SE}(\bar{x})=z^*\frac{\sigma}{\sqrt{n}}$$

The resulting confidence interval is
$$\boxed{\bar{x}\pm z^*\frac{\sigma}{\sqrt{n}}}$$
\end{frame}

% SLIDE 18
\begin{frame}{Precision, Confidence and Interval Width}
The formula shows how each component affects precision.
\begin{itemize}
  \item Increasing the sample size $n$ reduces the standard error and produces a narrower interval
  \item Increasing the confidence level increases $z^*$ and produces a wider interval
  \item Greater population variability $\sigma$ produces a wider interval
\end{itemize}
\begin{block}{The Confidence--Precision Trade-off}
For the same data, greater confidence requires a wider interval. A narrower interval provides greater precision but, without increasing the sample size, it must use a lower confidence level.
\end{block}

Because the margin of error depends on $\sqrt{n}$, multiplying the sample size by $4$ reduces the margin of error by one-half.
\end{frame}

% SECTION 4
\section{Interpreting Confidence Intervals}

% SLIDE 19

\begin{frame}{How to Interpret a Confidence Interval}
\begin{block}{Suitable Interpretation}
We are $95\%$ confident that the unknown population parameter lies between the reported endpoints. The confidence comes from using a procedure that captures the true parameter in approximately $95\%$ of repeated random samples when the required conditions are satisfied.
\end{block}

\begin{alertblock}{Common Misinterpretations}
A $95\%$ confidence interval does not mean that:
\begin{itemize}
  	\item There is a $95\%$ probability that the fixed parameter lies in this particular interval
  	\item $95\%$ of individual observations lie between the endpoints
 	 \item $95\%$ of future observations will lie between the endpoints
\end{itemize}
\end{alertblock}
\end{frame}

\section{Confidence Intervals for the Mean}

% SLIDE 21
\begin{frame}{The Four-Step Process for Confidence Intervals}
A consistent process keeps the statistical calculations connected to the scientific question.
\begin{enumerate}
	\item \textbf{STATE}

  	Identify the population, the parameter being estimated and the desired confidence level.
	\item \textbf{PLAN}

	Select an appropriate confidence-interval procedure and assess whether its conditions are reasonably satisfied.
	\item \textbf{SOLVE}

	Calculate the estimate, standard error, critical value, margin of error and confidence interval.
	\item \textbf{CONCLUDE}

	Interpret the interval in the context of the population and acknowledge any important limitations.
\end{enumerate}
\end{frame}

% SLIDE 22
\begin{frame}{Conditions for a Confidence Interval for $\mu$}
Fo one sample $z$ confidence interval we have these conditions.
\begin{block}{Conditions}
\begin{itemize}
	\item \textbf{Random Sampling}: The data should come from a random sample or another sampling process that reasonably represents the population of interest.
	\item \textbf{Independence}: The observations should be independent. When sampling without replacement, the sample size should generally be no more than $10\%$ of the population $n\leq0.10N$.
	\item \textbf{Approximately Normal Sampling Distribution}: The population should be approximately Normal, or the sample size should be sufficiently large for the Central Limit Theorem to apply. Strong skewness and extreme outliers require additional caution.
	\item \textbf{Known Standard Devation}: For the $z$ procedure (introduced here), the population standard deviation $\sigma$ is also treated as known.
\end{itemize}
\end{block}
\end{frame}

The population standard deviation is treated as known at $\sigma=15$ cm$^2$. What is a $95\%$ confidence interval for the population mean leaf surface area?

\end{block}

\begin{block}{STATE}
The parameter $\mu$ is the mean surface area, in cm$^2$, of all mature common bean leaves in this greenhouse population.
\end{block}
\end{frame}

% SLIDE 25
\begin{frame}{Example 14.3: Planning and Checking Conditions}
\begin{block}{PLAN}
A one sample $z$ confidence interval for $\mu$ will be used because the population standard deviation is treated as known.
\end{block}
\begin{itemize}
	\item \textbf{Random sampling:} The $31$ leaves were selected using a simple random sample
	\item \textbf{Independence:} The greenhouse contains more than $310$ mature leaves, so the sample contains less than $10\%$ of the population $31\leq0.10N$
	\item \textbf{Shape and sample size:} The sample distribution is somewhat left-skewed but has no extreme outliers. With $n=31$, the Central Limit Theorem provides reasonable support for an approximately Normal sampling distribution of $\bar{x}$
	\item \textbf{Known population standard deviation:}
	$$\sigma=15\text{ cm}^2$$
\end{itemize}
\end{frame}

% SLIDE 26
\begin{frame}{Example 14.3: Calculating the Interval}
\begin{block}{SOLVE}
The sample mean is
$$\bar{x}=\frac{3281}{31}\approx105.84\text{ cm}^2$$
The standard error is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}=\frac{15}{\sqrt{31}}\approx2.694\text{ cm}^2$$
For a $95\%$ confidence interval
$$z^*=1.960$$
The margin of error is
$$m=1.960(2.694)\approx5.28\text{ cm}^2$$
\end{block}
\end{frame}

% SLIDE 27
\begin{frame}{Example 14.3: Interval and Conclusion}
The confidence interval is
$$\bar{x}\pm m$$
$$105.84\pm5.28$$
Therefore
$$(100.56,\ 111.12)\text{ cm}^2$$

\begin{block}{CONCLUDE}
We are $95\%$ confident that the mean surface area of all mature common bean leaves in this greenhouse population is between approximately $100.6$ cm$^2$ and $111.1$ cm$^2$.
\end{block}
\end{frame}

\begin{frame}[fragile]{Example 14.3: Confidence Interval in R}
\scriptsize
\begin{verbatim}
# Observed leaf surface areas in square centimetres
leaf_area <- c(114, 100, 104, 89, 102, 91, 114, 114, 103, 105, 108, 130, 120, 132, 111, 128,
  118, 119, 86, 72, 111, 103, 74, 112, 107, 103, 98, 96, 112, 112, 93
)

# Population standard deviation
sigma <- 15

# Sample information, critical value and margin of error
x_bar <- mean(leaf_area)
n <- length(leaf_area)
standard_error <- sigma / sqrt(n)
z_star <- qnorm(0.975)
margin_error <- z_star * standard_error

# Calculate and display the confidence interval
confidence_interval <- c(lower = x_bar - margin_error, upper = x_bar + margin_error)
round(confidence_interval, 2)

> lower  upper
> 100.56 111.12
\end{verbatim}
\end{frame}

\vspace{0.2cm}
In most real studies, the population standard deviation is unknown.

\vspace{0.2cm}

\section{Hypothesis Testing}

% SLIDE 30
\begin{frame}{From Estimation to Hypothesis Testing}
A confidence interval begins with an unknown population parameter and asks which values are plausible based on the sample data.

\vspace{0.2cm}
A hypothesis test begins with a specific claim about a population parameter and asks whether the observed data are compatible with that claim.

\vspace{0.25cm}
\begin{block}{Central Question}
\centering
\large
If the population claim were true, how unusual would the observed sample result be?
\end{block}

\vspace{0.2cm}
A result that would be very unusual under the stated claim provides evidence against that claim. A result that is not unusual does not provide strong evidence against it.
\end{frame}

% SLIDE 31
\begin{frame}{The Null and Alternative Hypotheses}
A hypothesis test compares two competing statements about a population parameter.

\vspace{0.10cm}
\begin{block}{Null Hypothesis}
The \textbf{null hypothesis}, denoted by $H_0$, gives the specific population value or model being evaluated.
$$H_0:\mu=\mu_0$$
\end{block}

\begin{block}{Alternative Hypothesis}
The \textbf{alternative hypothesis}, denoted by $H_a$, describes the competing population claim for which evidence is being sought.
\end{block}

\vspace{0.1cm}
The test begins by temporarily assuming that $H_0$ is true. The observed data are then compared with what would be expected under this assumption.
\end{frame}


% SLIDE 32
\begin{frame}{One-Sided and Two-Sided Alternatives}
The form of the alternative hypothesis depends on the scientific question.
\begin{block}{Types of Hypothesis Tests}
\begin{itemize}
	\item \textbf{Lower-Tail Test}: A decrease or value below the reference level is being investigated.
	$$H_a:\mu<\mu_0$$
	\item \textbf{Upper-Tail Test}: An increase or value above the reference level is being investigated.
$$H_a:\mu>\mu_0$$
	\item \textbf{Two-Sided Test}: A difference in either direction is being investigated.
$$H_a:\mu\ne\mu_0$$
\end{itemize}
\end{block}
\end{frame}

% SLIDE 33
\begin{frame}{Type I and Type II Errors}
A hypothesis test uses sample data, so an incorrect decision is possible.
\begin{center}
\begin{tabular}{lcc}
\toprule
 & \textbf{$H_0$ is true} & \textbf{$H_0$ is false} \\
\midrule
\textbf{Reject $H_0$}
& Type I error
& Correct decision \\[1mm]

\textbf{Fail to reject $H_0$}
& Correct decision
& Type II error \\
\bottomrule
\end{tabular}
\end{center}

\begin{itemize}
	\item A \textbf{Type I error} occurs when a true null hypothesis is rejected. Its probability is denoted by $\alpha$
	\item A \textbf{Type II error} occurs when a false null hypothesis is not rejected. Its probability is denoted by $\beta$
\end{itemize}

\vspace{0.15cm}
The consequences of the two errors depend on the context. It is also important to say \textbf{fail to reject $H_0$}, rather than \textbf{accept $H_0$}, because the data may simply provide insufficient evidence against it.
\end{frame}

Researchers suspect that the population mean may be lower among adults aged $75$ to $79$. They randomly select $12$ patient records from a large health system population in this age group and record one phosphorus measurement from each patient. Assume that the population standard deviation remains $\sigma=0.1$ mmol/L. Does the sample provide evidence that the population mean phosphorus level for this older group is below $1.2$ mmol/L?
\end{block}

\small
\begin{center}
\begin{tabular}{rrrrrr}
\toprule

0.87 & 1.23 & 1.19 & 1.29 & 1.03 & 1.18 \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

% SLIDE 35
\begin{frame}{Examples 14.4 and 14.5: Phosphorus Levels}
The sample mean phosphorus level is
$$\bar{x}=\frac{13.54}{12}\approx1.1283\text{ mmol/L}$$
The observed difference from the reference value is
$$\bar{x}-\mu_0=1.1283-1.2=-0.0717\text{ mmol/L}$$
The sample mean is below $1.2$ mmol/L. However, a difference between $\bar{x}$ and $\mu$ can occur because of random sampling variation.

\begin{block}{Question for the Test}
If the population mean were actually $1.2$ mmol/L, how unusual would it be to obtain a sample mean of $1.1283$ mmol/L or lower from a random sample of $12$ patients?
\end{block}
\end{frame}

% SLIDE 36
\begin{frame}{Examples 14.4 and 14.5: Phosphorus Levels}
Let $\mu$ represent the population mean phosphorus level among adults aged $75$ to $79$ in the health-system population.

\vspace{0.2cm}
The null hypothesis states that the population mean equals the general-adult reference value:
$$H_0:\mu=1.2\text{ mmol/L}$$
The alternative hypothesis represents the researchers' suspicion that the population mean is lower:
$$H_a:\mu<1.2\text{ mmol/L}$$
This is a \textbf{lower-tail test} because only values below $1.2$ mmol/L support the alternative hypothesis.
\end{frame}

% SLIDE 37
\begin{frame}{Example 14.6: Aspirin Tablet Manufacturing}
\begin{block}{Question}
A pharmaceutical manufacturing process is designed to produce aspirin tablets with a population mean active ingredient amount of $325$ mg. Underfilling may reduce the effectiveness of a tablet, while overfilling may increase the risk of adverse effects and violate manufacturing specifications. What hypotheses should be used to evaluate whether the population mean active-ingredient amount differs from the target?
\end{block}

\vspace{0.2cm}
Let $\mu$ represent the population mean active ingredient amount for all tablets produced by the manufacturing process.

\vspace{0.2cm}
Because deviations above or below the target are both relevant, the appropriate test must consider both directions.
\end{frame}

% SLIDE 38
\begin{frame}{Example 14.6: Tablet Manufacturing}
The null hypothesis states that the manufacturing process is centred at the target amount:
$$H_0:\mu=325\text{ mg}$$
The alternative hypothesis states that the population mean differs from the target:
$$H_a:\mu\ne325\text{ mg}$$

\begin{block}{Interpretation}
This is a \textbf{two-sided test} because both underfilling and overfilling are departures from the intended manufacturing target.
\end{block}

\vspace{0.15cm}
The hypotheses refer to the population mean $\mu$, not to the sample mean $\bar{x}$. The sample mean will be used as evidence when evaluating the hypotheses.
\end{frame}

% SECTION 7
\section{P-Values and Statistical Significance}

% SLIDE 40
\begin{frame}{The Logic of a Hypothesis Test}
A hypothesis test evaluates the observed sample data under the temporary assumption that the null hypothesis is true.

\vspace{0.2cm}
The reasoning follows a consistent sequence:
\begin{enumerate}  
	\item Assume that $H_0$ is true
  	\item Determine the sampling distribution expected under $H_0$
  	\item Compare the observed statistic with this null distribution
	 \item Calculate the probability of obtaining a result at least as extreme as the observed result
  	\item Use this probability to evaluate the evidence against $H_0$
\end{enumerate}

\begin{block}{Central Idea}
A result that would be unusual when $H_0$ is true provides evidence against $H_0$.
\end{block}
\end{frame}

% SLIDE 41
\begin{frame}{The Test Statistic}
For a one-sample $z$ test of a population mean with known $\sigma$, the test statistic is
$$z=\frac{\bar{x}-\mu_0}{\sigma/\sqrt{n}}$$
Each part of the formula has a specific role:
\begin{itemize}
	\item $\bar{x}$ is the observed sample mean
	\item $\mu_0$ is the population mean stated by the null hypothesis
	\item $\bar{x}-\mu_0$ is the observed difference from the null value
\item $\sigma/\sqrt{n}$ is the standard error expected under the null model
\end{itemize}

\vspace{0.15cm}
The resulting $z$ statistic measures how many standard errors the observed sample mean is above or below the null value.
\begin{itemize}
	\item A positive $z$ indicates that $\bar{x}>\mu_0$
	\item A negative $z$ indicates that $\bar{x}<\mu_0$
	\item A value far from $0$ represents stronger disagreement with $H_0$
\end{itemize}
\end{frame}

% SLIDE 42
\begin{frame}{The P-Value}
\begin{block}{Definition}
The \textbf{P-value} is the probability of obtaining a test statistic at least as extreme as the observed statistic in the direction or directions specified by the alternative hypothesis. This probability is calculated under the assumption that the null hypothesis and model conditions are true.
\end{block}

\vspace{0.2cm}
The phrase \textbf{at least as extreme} includes the observed result and results that would provide even stronger evidence against $H_0$.

\vspace{0.2cm}
The P-value is calculated from the null sampling distribution. It therefore answers the following question:
\begin{center}
\textit{How unusual is the observed result if the null hypothesis is true?}
\end{center}
\end{frame}

% SLIDE 43
\begin{frame}{The Alternative Hypothesis Determines the Tail}
The direction of the alternative hypothesis determines which area of the null distribution is used to calculate the P-value.

\vspace{0.2cm}
\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Alternative} & \textbf{Type of test} & \textbf{P-value} \\
\midrule
$H_a:\mu<\mu_0$
& Lower-tail
& $P(Z\leq z_{\mathrm{obs}})$ \\[1mm]

$H_a:\mu>\mu_0$
& Upper-tail
& $P(Z\geq z_{\mathrm{obs}})$ \\[1mm]

$H_a:\mu\ne\mu_0$
& Two-sided
& $P(|Z|\geq|z_{\mathrm{obs}}|)$ \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.2cm}
For a symmetric Standard Normal distribution, the two-sided P-value can be calculated as
$$2P(Z\geq|z_{\mathrm{obs}}|)$$
\end{frame}

% SLIDE 44
\begin{frame}{Interpreting P-Values}
\begin{itemize}
	\item A \textbf{small P-value} indicates that the observed result would be unusual if $H_0$ were true. It therefore provides stronger evidence against $H_0$
	\item A \textbf{large P-value} indicates that the observed result remains reasonably compatible with $H_0$
	\item A large P-value does not prove that $H_0$ is true. The study may have insufficient evidence to detect a meaningful departure from $H_0$
\end{itemize}

\vspace{0.15cm}
\begin{alertblock}{What a P-Value Is Not}
A P-value is not
\begin{itemize}
	\item The probability that $H_0$ is true
	\item The probability that $H_a$ is true
	\item The probability that the observed result occurred only because of chance
  	\item A measure of the practical, clinical or biological importance of an effect
\end{itemize}
\end{alertblock}
\end{frame}

% SLIDE 45
\begin{frame}{Example 14.7: One-Sided Phosphorus Test}
For the phosphorus study, let $\mu$ represent the population mean phosphorus level among adults aged $75$ to $79$. The hypotheses are
$$H_0:\mu=1.2\text{ mmol/L}$$
$$H_a:\mu<1.2\text{ mmol/L}$$
The sample information is
$$n=12 \qquad \bar{x}\approx1.1283\text{ mmol/L} \qquad \sigma=0.1\text{ mmol/L}$$

\begin{block}{Question}
Assuming that $H_0$ is true, what is the probability of obtaining a sample mean of $1.1283$ mmol/L or lower from a random sample of $12$ patients?
\end{block}
\end{frame}

% SLIDE 46
\begin{frame}{Example 14.7: The Null Sampling Distribution}
Under the null hypothesis, the sampling distribution of $\bar{x}$ is centred at the null value
$$\mu_{\bar{x}}=\mu_0=1.2\text{ mmol/L}$$

The standard error is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}=\frac{0.1}{\sqrt{12}}\approx0.0289\text{ mmol/L}$$
Therefore, under $H_0$, sample means are expected to vary around $1.2$ mmol/L with a standard error of approximately $0.0289$ mmol/L.

\vspace{0.2cm}
The next step is to express the observed difference between $\bar{x}$ and $\mu_0$ in standard error units.
\end{frame}

% SLIDE 47
\begin{frame}{Example 14.7: Calculating the Test Statistic}
The observed difference from the null value is
$$\bar{x}-\mu_0=1.1283-1.2=-0.0717\text{ mmol/L}$$
The test statistic is
$$z_{\mathrm{obs}}=\frac{\bar{x}-\mu_0}{\sigma/\sqrt{n}}$$
$$z_{\mathrm{obs}}=\frac{1.1283-1.2}{0.1/\sqrt{12}}\approx-2.48$$
The observed sample mean is approximately $2.48$ standard errors below the mean specified by $H_0$. Because the alternative hypothesis is
$$H_a:\mu<1.2$$
the P-value is the Standard Normal area to the left of $z=-2.48$.
\end{frame}

% SLIDE 48
\begin{frame}{Example 14.7: P-Value and Conclusion}
The one-sided P-value is
$$P(Z\leq-2.48)=N(-2.48)=1-N(2.48)=1-0.9934=0.0066$$
\begin{block}{Interpretation of the P-Value}
If the population mean phosphorus level were $1.2$ mmol/L, the probability of obtaining a sample mean of approximately $1.1283$ mmol/L or lower from a random sample of $12$ patients would be about $0.0066$ or $0.66\%$.
\end{block}
\begin{block}{Conclusion}
The data provide strong evidence against $H_0$ and support the conclusion that the population mean phosphorus level among adults aged $75$ to $79$ in the health-system population is below $1.2$ mmol/L, provided that the model conditions are reasonably satisfied.
\end{block}
\end{frame}

% SLIDE 49
\begin{frame}[fragile]{Example 14.7: One-Sided Test in R}
\scriptsize
\begin{verbatim}
# Observed phosphorus measurements in mmol/L
phosphorus <- c(1.26, 1.39, 1.00, 1.00, 1.00, 1.10, 0.87, 1.23, 1.19, 1.29, 1.03, 1.18)

# Null value and known population standard deviation
mu_0 <- 1.2
sigma <- 0.1

# Sample information
x_bar <- mean(phosphorus)
n <- length(phosphorus)
standard_error <- sigma / sqrt(n)

# Test statistic and p-value
z_observed <- (x_bar - mu_0) / standard_error
p_value <- pnorm(z_observed)

# Display the results
results <- c(sample_mean = x_bar, standard_error = standard_error,
  z_statistic = z_observed, p_value = p_value)

 sample_mean standard_error     z_statistic          p_value
       1.1283         0.0289         -2.4826           0.0065
\end{verbatim}
\end{frame}

% SLIDE 50
\begin{frame}{Example 14.8: Two-Sided Aspirin Test}
\begin{block}{Question}
A manufacturing process targets $\mu=325$ mg with known $\sigma=5$ mg. A random sample of $n=10$ tablets has mean $\bar{x}=326.9$ mg.
The hypotheses are
$$H_0:\mu=325$$
$$H_a:\mu\ne325$$

What is the two-sided P-value?
\end{block}
\end{frame}

% SLIDE 51
\begin{frame}{Example 14.8: Standard Error and Test Statistic}
Under $H_0$, the sampling distribution of $\bar{x}$ is centred at $325$ mg. Its standard error is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}=\frac{5}{\sqrt{10}}\approx1.581\text{ mg}$$
The test statistic is
$$z_{\mathrm{obs}}=\frac{\bar{x}-\mu_0}{\sigma/\sqrt{n}}=\frac{326.9-325}{1.581}\approx1.202$$
The sample mean is approximately $1.202$ standard errors above the target value. Because the alternative hypothesis is two-sided, equally extreme results above or below $325$ mg must be included in the P-value.
\end{frame}

% SLIDE 52
\begin{frame}{Example 14.8: Standard Error and Test Statistic}
For a two-sided test, the p-value includes both tails of the Standard Normal distribution:
$$P(|Z|\geq1.202)=P(Z\leq-1.202)+P(Z\geq1.202)=N(-1.202)+(1-N(1.202))\approx0.2295$$

\begin{itemize}
	\item If the population mean active ingredient amount were $325$ mg, there would be approximately a $22.95\%$ probability of obtaining a sample mean at least as far from $325$ mg as $326.9$ mg in either direction.
	\item Since
	$$0.2295>0.05$$ 
	the result is not statistically significant at the $\alpha=0.05$ significance level. We fail to reject $H_0$.
	\item The sample does not provide sufficient evidence that the population mean active-ingredient amount differs from $325$ mg. This conclusion does not prove that the process mean is exactly $325$ mg.
\end{itemize}
\end{frame}

% SLIDE 53
\begin{frame}[fragile]{Example 14.8: Standard Error and Test Statistic}
\scriptsize
\begin{verbatim}
# Null value and known population standard deviation
mu_0 <- 325
sigma <- 5

# Sample information
n <- 10
x_bar <- 326.9

# Standard error and test statistic
standard_error <- sigma / sqrt(n)
z_observed <- (x_bar - mu_0) / standard_error

# Two-sided P-value
# abs() measures distance from zero in either direction.
upper_tail <- 1 - pnorm(abs(z_observed))
p_value <- 2 * upper_tail

# Display the results
results <- c(standard_error = standard_error, z_statistic = z_observed, p_value = p_value)
round(results, 4)
standard_error     z_statistic          p_value
        1.5811          1.2017           0.2295
\end{verbatim}
\end{frame}

% SECTION 8
\section{Significance Levels and One-Sample Tests}

% SLIDE 54
\begin{frame}{The Significance Level $\alpha$}
The \textbf{significance level}, denoted by $\alpha$, is the decision threshold chosen before examining the sample results.

\vspace{0.2cm}

The p-value is compared with $\alpha$:
\begin{itemize}
	\item If $P\leq\alpha$, the result is \textbf{statistically significant} and $H_0$ is rejected
  	\item If $P>\alpha$, the result is \textbf{not statistically significant} and we fail to reject $H_0$
\end{itemize}
Common significance levels include
$$\alpha=0.05$$
and
$$\alpha=0.01$$
The value of $\alpha$ should reflect the context and the consequences of making a Type I error. It should not be selected after seeing the p-value.
\end{frame}

% SLIDE 55
\begin{frame}{What Does $\alpha$ Mean?}
For a correctly calibrated hypothesis test, $\alpha$ controls the long-run probability of rejecting $H_0$ when $H_0$ is true.
$$\alpha=P(\text{reject }H_0\mid H_0\text{ is true})$$
Therefore
$$\alpha=0.05$$
means that, over many repetitions in which $H_0$ is true, approximately $5\%$ of the tests would incorrectly reject it.
\end{frame}

% SLIDE 56
\begin{frame}{The One-Sample $z$ Test for a Mean}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.46\textwidth}
The null hypothesis is
$$H_0:\mu=\mu_0$$
The test statistic is
$$z=\frac{\bar{x}-\mu_0}{\sigma/\sqrt{n}}$$
The statistic measures the observed difference from the null value in standard error units.
\end{column}

\begin{column}{0.50\textwidth}
\textbf{Conditions}
\begin{itemize}
	\item The data come from a random or reasonably representative sample
	\item The observations are independent
	\item The population is Normal or the sample is sufficiently large for the Central Limit Theorem
	\item The population standard deviation $\sigma$ is treated as known
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% SLIDE 59
\begin{frame}{Example 14.9: Body Temperature}
\begin{block}{Question}
An historical reference value for mean human body temperature is $98.6^\circ$F. A later study records one oral temperature from each of $n=130$ healthy adult volunteers. The sample mean is
$$\bar{x}=98.25^\circ\text{F}$$
Assume that the observations are independent and reasonably representative of the population of interest. Also assume that the population standard deviation is known to be
$$\sigma=0.6^\circ\text{F}$$
At the $\alpha=0.05$ significance level, do the data provide evidence that the population mean body temperature differs from $98.6^\circ$F?
\end{block}
\end{frame}

% SLIDE 60
\begin{frame}{Example 14.9: State and Plan}
Let $\mu$ represent the population mean oral temperature of healthy adults in the population of interest.
\vspace{0.15cm}
\begin{block}{STATE}
The question concerns a difference in either direction, so the hypotheses are
$$H_0:\mu=98.6^\circ\text{F}$$
$$H_a:\mu\ne98.6^\circ\text{F}$$
The significance level is $\alpha=0.05$.
\end{block}
\begin{block}{PLAN}
A one-sample $z$ test for a population mean will be used.
The observations are assumed to be independent and representative. The sample size is large and the population standard deviation is treated as known.
\end{block}
\end{frame}

% SLIDE 61
\begin{frame}{Example 14.9: Standard Error and Test Statistic}
The standard error is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}=\frac{0.6}{\sqrt{130}}\approx0.0526^\circ\text{F}$$
The observed difference from the null mean is
$$\bar{x}-\mu_0=98.25-98.6=-0.35^\circ\text{F}$$
The test statistic is
$$z_{\mathrm{obs}}=\frac{98.25-98.6}{0.6/\sqrt{130}}\approx-6.65$$
The observed sample mean is approximately $6.65$ standard errors below the value specified by $H_0$.
\end{frame}

% SLIDE 62
\begin{frame}{Example 14.9: P-Value and Conclusion}
Because the alternative hypothesis is two-sided, the p-value includes extreme results in both tails
$$P=2P(Z\leq-6.65)\approx2.9\times10^{-11}$$
The p-value is much smaller than the significance level
$$2.9\times10^{-11}<0.05$$
Therefore, we reject $H_0$.

\begin{block}{Conclusion}
The data provide extremely strong evidence that the population mean oral temperature of healthy adults in the population of interest differs from $98.6^\circ$F. The sample result suggests that the population mean is lower than the historical reference value.
\end{block}
\end{frame}

% SLIDE 63
\begin{frame}[fragile]{Example 14.9: Body Temperature Test in R}
\scriptsize
\begin{verbatim}
# Null value and known population standard deviation
mu_0 <- 98.6
sigma <- 0.6

# Sample information
x_bar <- 98.25
n <- 130

# Standard error and test statistic
standard_error <- sigma / sqrt(n)
z_observed <- (x_bar - mu_0) / standard_error

# Two-sided P-value
p_value <- 2 * pnorm(-abs(z_observed))

# Display the results
results <- c(standard_error = standard_error, z_statistic = z_observed, p_value = p_value)
signif(results, 4)

\end{verbatim}
\end{frame}

% SECTION 9
\section{Tests from Confidence Intervals}

% SLIDE 64
\begin{frame}{The Confidence Interval and Test Connection}
Two-sided confidence intervals and two-sided hypothesis tests are closely connected when they use the same statistical model and standard error.

\vspace{0.2cm}
A two-sided hypothesis test with significance level $\alpha$ corresponds to a confidence interval with confidence level
$$C=1-\alpha$$

\begin{center}
\begin{tabular}{cc}
\toprule
\textbf{Significance level $\alpha$} & \textbf{Confidence level $C$} \\
\midrule
$0.10$ & $90\%$ \\
$0.05$ & $95\%$ \\
$0.01$ & $99\%$ \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.15cm}
The confidence interval identifies the parameter values that remain reasonably compatible with the sample data. The hypothesis test evaluates one particular value from that set of possibilities.
\end{frame}

% SLIDE 65
\begin{frame}{Using a Confidence Interval to Test a Null Value}
Consider a two-sided hypothesis test
$$H_0:\mu=\mu_0$$
using significance level $\alpha$. The corresponding $(1-\alpha)100\%$ confidence interval leads to the same decision:

\vspace{0.15cm}
\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Location of $\mu_0$} & \textbf{Test decision} & \textbf{P-value} \\
\midrule
Inside the interval
& Fail to reject $H_0$
& $P>\alpha$ \\[1mm]

Outside the interval
& Reject $H_0$
& $P<\alpha$ \\
\bottomrule
\end{tabular}
\end{center}
\begin{block}{Equivalent Interpretation}
The confidence interval contains the parameter values that would not be rejected by the corresponding two-sided hypothesis test.
\end{block}
\end{frame}

% SLIDE 66
\begin{frame}{Example 14.10: Body Temperature Confidence Interval}
For the body temperature study, the sample information is
$$\bar{x}=98.25^\circ\text{F} \quad n=130 \quad \sigma=0.6^\circ\text{F}$$
\begin{block}{Question}
What is the $95\%$ confidence interval for the population mean oral temperature of healthy adults in the population of interest? How does this interval lead to the same conclusion as the two-sided test
$$H_0:\mu=98.6^\circ\text{F}$$
at the $\alpha=0.05$ significance level?
\end{block}
\end{frame}

% SLIDE 67
\begin{frame}{Example 14.10: Calculating the Interval}
For a $95\%$ confidence interval, the Standard Normal critical value is
$$z^*=1.960$$
The standard error is
$$\operatorname{SE}(\bar{x})=\frac{\sigma}{\sqrt{n}}=\frac{0.6}{\sqrt{130}}\approx0.0526^\circ\text{F}$$
The margin of error is
$$m=z^*\operatorname{SE}(\bar{x})$$
$$m=1.960(0.0526)\approx0.1031^\circ\text{F}$$
Therefore, the interval extends approximately $0.1031^\circ$F on each side of the sample mean.
\end{frame}


% SLIDE 68
\begin{frame}{Example 14.10: Interval and Test Conclusion}
The $95\%$ confidence interval is
$$\bar{x}\pm m$$
$$98.25\pm0.1031$$
Therefore
$$\boxed{(98.15,\ 98.35)^\circ\text{F}}$$

\begin{block}{Confidence-Interval Interpretation}
We are $95\%$ confident that the population mean oral temperature of healthy adults in the population of interest is between approximately $98.15^\circ$F and $98.35^\circ$F.
\end{block}
The null value is outside the interval because
$$98.6>98.35$$
Therefore, the corresponding two-sided test rejects $H_0$ at $\alpha=0.05$. Both procedures provide evidence that the population mean differs from $98.6^\circ$F.
\end{frame}


% SLIDE 69
\begin{frame}[fragile]{Example 14.10: Confidence Interval in R}
\scriptsize
\begin{verbatim}
# Sample information
x_bar <- 98.25
n <- 130

# Known population standard deviation
sigma <- 0.6

# Standard error and 95% critical value
standard_error <- sigma / sqrt(n)
z_star <- qnorm(0.975)

# Margin of error
margin_error <- z_star * standard_error

# Confidence interval
confidence_interval <- c(lower = x_bar - margin_error, upper = x_bar + margin_error)
round(confidence_interval, 2)

lower upper
98.15 98.35
\end{verbatim}
\end{frame}
