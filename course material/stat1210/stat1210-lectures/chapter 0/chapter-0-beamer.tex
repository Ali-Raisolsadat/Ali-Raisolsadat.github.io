\begin{block}{Course Goal}
Develop the statistical reasoning needed to organize data, describe patterns, quantify uncertainty and draw appropriate conclusions in life-science and social-science settings.
\end{block}

%------------------------------------------------------------
% Course roadmap I and II
%------------------------------------------------------------

\begin{frame}{Course Roadmap: Exploring Data and Probability}

\large In the first part of the course, we begin by learning how to describe data and using probability to model and understand uncertainty.

  \begin{block}{I. Exploring Data}
    \begin{itemize}
      \item \textbf{Chapters 1--2:} Distributions and numerical summaries
      \item \textbf{Chapter 3:} Scatterplots and correlation
      \item \textbf{Chapter 4:} Regression
    \end{itemize}
  \end{block}

  \begin{block}{II. Probability and Distributions}
    \begin{itemize}
      \item \textbf{Chapters 9--10:} Probability rules
      \item \textbf{Chapter 11:} Normal distributions
      \item \textbf{Chapter 12:} Binomial and Poisson distributions
    \end{itemize}
  \end{block}

\end{frame}

%------------------------------------------------------------
% Course roadmap III and IV
%------------------------------------------------------------

\begin{frame}{Course Roadmap: Statistical Inference}

\large In the second half of the course we use sample data to make conclusions about populations.

  \begin{block}{III. Foundations of Inference}
    \begin{itemize}
      \item \textbf{Chapter 6:} Samples and observational studies
      \item \textbf{Chapter 13:} Sampling distributions
      \item \textbf{Chapter 14:} Confidence intervals and $p$-values
      \item \textbf{Chapter 15:} Inference in practice
    \end{itemize}
  \end{block}

  \begin{block}{IV. Specific Procedures}
    \begin{itemize}
      \item \textbf{Chapter 17:} Inference about a population mean
      \item \textbf{Chapter 19:} Inference about a population proportion
    \end{itemize}
  \end{block}

\end{frame}
